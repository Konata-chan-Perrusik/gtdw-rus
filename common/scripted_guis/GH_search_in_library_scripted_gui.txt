gh_can_read_more_sgui = {
	saved_scopes = { gh_book_name_flag }
	is_valid = {
		trigger_if = {
			limit = {
				has_variable = gh_number_of_recently_read_book
				has_variable = gh_max_number_of_book_read
			}
			var:gh_number_of_recently_read_book < var:gh_max_number_of_book_read
		}
		trigger_else = {}
	}
}

gh_read_book_sgui = {
	effect = {
		scope:character_scope ?= {
			add_to_variable_list = {
				name = gh_book_personally_read
				target = root
			}
			if = {
				limit = {
					NOT = { has_variable = gh_number_of_recently_read_book }
				}
				set_variable = {
					name = gh_number_of_recently_read_book
					value = 1
					years = 1
				}
			}
			else = {
				change_variable = {
					name = gh_number_of_recently_read_book
					add = 1
				}
			}
			if = {
				limit = {
					NOT = { has_variable = gh_has_read_a_book_in_session }
				}
				set_variable = {
					name = gh_has_read_a_book_in_session
					value = yes
				}
			}
			random_list = {
				70 = {} # No Bonus
				30 = {  # Temporary/Permanent Bonus
					modifier = {
						scope:gh_rarity_of_book ?= flag:masterwork
						add = 5
					}
					modifier = {
						scope:gh_rarity_of_book ?= flag:famed
						add = 7
					}
					modifier = {
						scope:gh_rarity_of_book ?= flag:illustrious
						add = 10
					}

					modifier = {
						stewardship <= low_skill_rating
						add = -5
					}
					modifier = {
						stewardship >= high_skill_rating
						add = 5
					}
					modifier = {
						stewardship >= very_high_skill_rating
						add = 5
					}
					gh_add_topic_book_bonus_effect = yes
				}
			}			
		}
	}
}

		exists = scope:gh_book_name_flag

			scope:book_story ?= { var:book_rarity ?= flag:common }

gh_buy_book_from_collection_sgui = {
	saved_scopes = { bought_book_name book_story }
	is_shown = {
		trigger_if = { # If the book has been deposited
			limit = { 
				exists = scope:book_story 
				scope:book_story ?= { has_variable = book_donor  }
			}
			NOT = { scope:book_story.var:book_donor = root  } #Then we make sure it's not the player trying to buy it
		}
		trigger_else = {}
	}
	effect = {
		set_variable = {
			name = gh_attempt_to_buy_book_in_collection
			value = scope:bought_book_name
		}
		
		# If the book has a story variable, we store it so we can later retrieve the procedural title and description saved in the local collection story under the book's name.
		if = {
			limit = { scope:book_story ?= { has_variable = gh_book_text_holder } }
			set_variable = {
				name = character_text_holder_var
				value = scope:book_story.var:gh_book_text_holder
			}
		}
		else = {
			remove_variable = character_text_holder_var
		}
		if = {
			limit = { scope:book_story ?= { has_variable = book_rarity } }
			set_variable = {
				name = gh_temporary_book_rarity
				value = scope:book_story.var:book_rarity
			}
		}
		if = {
			limit = { exists = scope:book_story }
			set_variable = {
				name = book_story_var
				value = scope:book_story
			}
		}

		open_interaction_window = {
			interaction = gh_buy_book_from_library_interaction
			actor = root
			recipient = root
		}
	}
}
