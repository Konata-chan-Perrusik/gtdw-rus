GetInspirationCategory = { # Localize whether the Inspiration is a Personal Inspiration
	type = character

	text = { # Personal Inspiration
		trigger = { exists = var:personal_inspiration_story }
		localization_key = inspiration_category_personal_inspiration
	}

	text = { localization_key = inspiration_category_regular_inspiration } # Inspiration (vanilla)
}

GetExpectedPersonalInspirationArtifactRarity = { # Get the expected Artifact Rarity in a Personal Inspiration
	type = character

	text = { # Common
		trigger = { var:personal_inspiration_story.personal_inspiration_combined_rarity_value < 50 }
		localization_key = expected_personal_inspiration_artifact_rarity_common
	}

	text = { # Masterwork
		trigger = { var:personal_inspiration_story.personal_inspiration_combined_rarity_value < 130 }
		localization_key = expected_personal_inspiration_artifact_rarity_masterwork
	}

	text = { # Famed
		trigger = { var:personal_inspiration_story.personal_inspiration_combined_rarity_value < 180 }
		localization_key = expected_personal_inspiration_artifact_rarity_famed
	}

	text = { localization_key = expected_personal_inspiration_artifact_rarity_illustrious } # Illustrious
}
