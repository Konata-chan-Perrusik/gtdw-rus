gh_buy_book_from_library_interaction = {
	category = interaction_category_friendly
	icon = icon_scheme_coerce_contribution
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:recipient = { NOT = { has_variable = character_text_holder_var } } }
				desc = gh_buy_book_from_library_interaction_desc
			}
			desc = gh_buy_book_from_library_interaction_desc_alt
		}
	}

	interface_priority = 80
	desc = gh_buy_book_from_library_interaction_desc
	common_interaction = yes
	
	is_shown = {
		OR = {
			scope:actor ?= { has_variable = library_interaction_visible }
			always = no
		}
	}
	
	is_valid_showing_failures_only = {}
	auto_accept = yes

	send_option = {
		flag = gh_purchase_book_directly
		localization = gh_purchase_book_directly_name
		current_description = gh_purchase_book_directly
		starts_enabled = { always = yes }
	}	
	send_option = {
		flag = gh_commission_book_copy
		localization = gh_commission_book_copy_name
		current_description = gh_commission_book_copy
		is_valid = {
			scope:actor = {
				custom_tooltip = {
					text = gh_is_already_buying_book_tt
					NOT = { has_character_flag = gh_is_already_buying_book }
				}
			}
		}
	}


	on_accept = {
		scope:recipient = {
			if = {
				limit = { NOT = { has_variable = character_text_holder_var } }
				custom_tooltip = gh_purchase_book_artifact_tooltip	
			}
			else = { custom_tooltip = gh_purchase_book_artifact_tooltip_alt	} 
			if = {
				limit = { scope:gh_purchase_book_directly = yes }			
				remove_short_term_gold = {
					value = gh_book_purchase_cost_value
					multiply = 1.5
				}
				if = {
					limit = { NOT = { has_variable = character_text_holder_var } }
					custom_tooltip = {
						text = gh_purchase_book_directly_tooltip
						location ?= { #ASKIL NOTE: If collections should be county-based, change value here
							var:gh_search_library_story ?= {
								gh_remove_book_from_collection_effect = yes
							}
						}
					}
				}
				else = { 
					custom_tooltip = {
						text = gh_purchase_book_directly_tooltip_alt
						location ?= { #ASKIL NOTE: If collections should be county-based, change value here
							var:gh_search_library_story ?= {
								gh_remove_book_from_collection_effect = yes
							}
						}
					}
				} 
				gh_create_book_artifact_effect = { BUYER = scope:recipient }
			}
			else_if = {
				limit = { scope:gh_commission_book_copy = yes }
				remove_short_term_gold = gh_book_purchase_cost_value
				if = {
					limit = { NOT = { has_variable = character_text_holder_var } }
					custom_tooltip = gh_commission_book_copy_tooltip
				}
				else = { custom_tooltip = gh_commission_book_copy_tooltip_alt	} 
				add_character_flag = {
					flag = gh_is_already_buying_book
					months = 7
				}
				trigger_event = {
					id = gh_search_in_library.0004
					months = 6
				}
			}
			remove_variable = character_text_holder_var
		}
	}
}
