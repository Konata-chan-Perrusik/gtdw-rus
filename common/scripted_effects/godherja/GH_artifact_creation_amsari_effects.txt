create_artifact_aushedar_armor_effect = {
	# Get the character the artifact is being made for.
	$OWNER$ = { save_scope_as = owner }
	
	set_artifact_rarity_illustrious = yes
	
	create_artifact = {
		name = artifact_armor_of_aushedar_name
		description = artifact_armor_of_aushedar_description		
		type = armor_plate
		visuals = armor
		wealth = scope:wealth
		quality = scope:quality
		modifier = artifact_prowess_4_modifier
		save_scope_as = newly_created_artifact
		history = { type = created_before_history }
	}
	
	scope:newly_created_artifact = {
		set_variable = { name = historical_unique_artifact value = yes }
		save_scope_as = epic
		add_artifact_title_history = {
			target = title:e_amsari_hiklahn #Amsar
			date = 6011.3.5
		}
	}
}
