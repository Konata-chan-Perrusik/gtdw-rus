gh_create_book_artifact_effect = {
	$BUYER$ = { save_scope_as = character_book_buyer }
	# If the book is historical, the process is much simpler
	if = {
		limit = {
			scope:character_book_buyer ?= { has_variable = gh_attempt_to_buy_book_in_collection }
			is_target_in_variable_list = { 
				name = gh_global_historical_book_list
				target = scope:character_book_buyer.var:gh_attempt_to_buy_book_in_collection
			} 	
		}
		gh_create_historical_book_from_library_effect = yes
	}
	else_if = {
		limit = { scope:character_book_buyer ?= { has_variable = gh_attempt_to_buy_book_in_collection } }
		# First we check the rarity of the book
		if = {
			limit = {
				OR = {
					scope:character_book_buyer.var:gh_temporary_book_rarity ?= flag:illustrious # Books created from procedural artifacts have their own variable for rarity
					any_in_global_list = {
						variable = global_illustrious_book_list
						this = scope:character_book_buyer.var:gh_attempt_to_buy_book_in_collection
					}
				}
			}
			set_artifact_rarity_illustrious = yes
		}
		else_if = {
			limit = {
				OR = {
					scope:character_book_buyer.var:gh_temporary_book_rarity ?= flag:famed
					any_in_global_list = {
						variable = global_famed_book_list
						this = scope:character_book_buyer.var:gh_attempt_to_buy_book_in_collection
					}
				}
			}
			set_artifact_rarity_famed = yes
		}
		else_if = {
			limit = {
				OR = {
					scope:character_book_buyer.var:gh_temporary_book_rarity ?= flag:masterwork
					any_in_global_list = {
						variable = global_masterwork_book_list
						this = scope:character_book_buyer.var:gh_attempt_to_buy_book_in_collection
					}
				}
			}
			set_artifact_rarity_masterwork = yes
		}
		else = {
			set_artifact_rarity_common = yes
		}
		scope:character_book_buyer ?= { remove_variable = gh_temporary_book_rarity }
		
		if = {
			limit = { 
				trigger_if = {
					limit = { scope:character_book_buyer ?= { has_variable = book_story_var } }
					scope:character_book_buyer.var:book_story_var ?= { 
						has_variable_list = GH_artifiact_modifier_list
					}
				}
				trigger_else = { always = no }
			}
			random_dummy_gender_effect = yes #We don't really have a creator so we just use a random one
			GH_create_artifact_book_light_effect = {
				OWNER = scope:actor
				CREATOR = scope:dummy_gender
				SET_SUBJECT = flag:no
				SET_TOPIC = flag:no
			}
		}	
		else = {
			random_dummy_gender_effect = yes
			create_artifact_book_effect = {
				OWNER = scope:actor
				CREATOR = scope:dummy_gender
				SET_SUBJECT = flag:no
				SET_TOPIC = flag:no
			}
		}

		hidden_effect_new_object = {
			scope:newly_created_artifact = {
				set_variable = { name = historical_book value = scope:character_book_buyer.var:gh_attempt_to_buy_book_in_collection }
				
				# If the book in the collection was previously a procedural artifact, its title and description are retrieved from another loc entry
				if = {
					limit = { scope:character_book_buyer = { has_variable = character_text_holder_var } }
					set_variable = { 
						name = gh_artifact_text_holder 
						value = scope:character_book_buyer.var:character_text_holder_var 
					}
					set_artifact_name = GH_PROCEDURAL_ARTIFACT_BOOK_TITLE					
					set_artifact_description = GH_PROCEDURAL_ARTIFACT_BOOK_DESC
				}
				# Otherwise we use the normal method to set the title and description
				else = {
					set_artifact_name = artifact_book_historical_title				
					set_artifact_description = artifact_book_historical_description
				}
				remove_variable = character_text_holder_var
		
				scope:character_book_buyer.var:book_story_var ?= {
					if = {
						limit = { has_variable_list = knowledge_entry_list }
						every_in_list = {
							variable = knowledge_entry_list
							scope:newly_created_artifact = {
								add_to_variable_list = {
									name = knowledge_entry_list
									target = prev
								}
							}
						}
					}
				}
			}
		}
		remove_variable = gh_attempt_to_buy_book_in_collection
	}
	if = {
		limit = { scope:character_book_buyer ?= { has_variable = book_story_var } }
		scope:character_book_buyer.var:book_story_var ?= {
			end_story = yes
		}
	}
}

build_collection_of_topics_effect = {
	save_scope_as = reader
	location.county.holder ?= { save_scope_as = gh_location_ruler }
	if = {
		limit = { NOT = { this.location = { has_variable = gh_search_library_story } } }
		create_story = { 
			type = gh_search_library_story
			save_scope_as = scoped_gh_search_library_story
		}
		scope:scoped_gh_search_library_story ?= {
			GH_create_library_topic_effect = { VALUE = gh_pre_creation_remaining_location_topic_capacity_value }
		}
	}
	else_if = {
		limit = { 
			this.location = { has_variable = gh_search_library_story } 
			NOT = { this.location.var:gh_search_library_story = { has_variable = recently_checked_collection  } }
			gh_post_creation_remaining_location_topic_capacity_value > 0
		}
		this.location.var:gh_search_library_story ?= {
			GH_create_library_topic_effect = { VALUE = gh_post_creation_remaining_location_topic_capacity_value }
		}
	}
}

transfer_book_to_library = {
	# If this is a historical Godherja™ book then we just need to add the correct flag
	if = {
		limit = {
			exists = scope:recipient.var:gh_transfer_book_to_collection.var:historical_book
			is_target_in_global_variable_list = {
				name = gh_global_historical_book_list
				target = scope:recipient.var:gh_transfer_book_to_collection.var:historical_book
			}
		}
		gh_transfer_historical_book_to_library_effect = { COLLECTOR = scope:recipient TOPIC = $TOPIC$ }
		destroy_artifact = scope:recipient.var:gh_transfer_book_to_collection
		remove_variable = gh_transfer_book_to_collection
	}
	# If it's not then it's a whole lot more work
	else = {
		gh_transfer_procedural_book_to_library = { COLLECTOR = scope:recipient TOPIC = $TOPIC$ }
	}
}

increase_global_book_rarity_variable_effect = {
	# Only a few book should become rare in each topics
	if = {
		limit = { NOT = { has_variable = book_rarity_set_var } }
		set_variable = { name = book_rarity_set_var value = 1 }
	}
	else = { change_variable = { name = book_rarity_set_var add = 1 } }
	# Only a few regional should become rare in the world
	if = {
		limit = { NOT = { has_global_variable = $REGION$_books_increased_rarity } }
		set_global_variable = { name = $REGION$_books_increased_rarity value = 1 }
	}
	else = { change_global_variable = { name = $REGION$_books_increased_rarity add = 1 } }
}

create_procedural_book_attribute_story = {
	scope:gh_collector_procedural = {
		# We create the story on the player
		create_story = {
			type = gh_book_attribute_story
			save_scope_as = book_attribute_story_scope
		}
		# We assume the player already has read the book
		add_to_variable_list = {
			name = gh_book_personally_read
			target = flag:$BOOK_FLAG$
		}
	}
	#Then we set the story as a variable inside the library story
	set_variable = {
		name = $BOOK_FLAG$
		value = scope:book_attribute_story_scope
	}	
	
	gh_add_book_to_local_collection_in_topic_effect = { GIVER = scope:gh_collector_procedural TOPIC = $TOPIC$  BOOK = $BOOK_FLAG$  }
	
	var:$BOOK_FLAG$ ?= {
		set_variable = {
			name = book_donor
			value = scope:gh_collector_procedural
		}
		
		# Set GH_artifiact_modifier_list variable list so we can retrieve the correct modifiers in the future
		GH_add_artifact_modifier_to_list_effect = yes
		
		# Set rarity of book for GUI Collection Stuff
		if = {
			limit = { scope:gh_collector_procedural.var:gh_transfer_book_to_collection = { rarity = masterwork } }
			set_variable = {
				name = book_rarity
				value = flag:masterwork
			}
		}
		else_if = {
			limit = { scope:gh_collector_procedural.var:gh_transfer_book_to_collection = { rarity = famed } }
			set_variable = {
				name = book_rarity
				value = flag:famed
			}
		}											
		else_if = {
			limit = { scope:gh_collector_procedural.var:gh_transfer_book_to_collection = { rarity = illustrious } }
			set_variable = {
				name = book_rarity
				value = flag:illustrious
			}
		}
		scope:gh_collector_procedural.var:gh_transfer_book_to_collection ?= {
			# Every knowledge entry on the artifact we add it to the book story
			every_in_list = {
				variable = knowledge_entry_list
				scope:gh_collector_procedural.location.var:gh_search_library_story.var:$BOOK_FLAG$ ?= { #ASKIL NOTE: If collections should be county-based, change value here
					add_to_variable_list = {
						name = knowledge_entry_list
						target = prev
					}
				}
			}
		}
		
		# We find a rando to hold the book artifact title and description
		random_ruler = {
			limit = { NOT = { has_variable = is_holding_text } }
			save_scope_as = random_text_holder
			
			# Make sure the character isn't picked again to hold text
			set_variable = {
				name = is_holding_text
				value = yes
			}
			
			# We save the rando so we can scope to them in GUI and display the title and desc of the book
			prev = {
				set_variable = {
					name = gh_book_text_holder
					value = scope:random_text_holder
				}
			}			
			scope:gh_collector_procedural = {
				# We save on a variable the rando to be re-used in gh_search_in_library.0003
				set_variable = {
					name = gh_book_text_holder
					value = scope:random_text_holder
				}
				trigger_event = gh_search_in_library.0003
			}
		}
	}
}

				get_book_content_quality_effect = { SKILL_BASE = scope:author }
			}
			# We save these values as variables on the owner so that we can access them deep in scopes in the desc-loc later on
			scope:owner = {
				set_variable = {
					name = quality
					value = scope:quality
				}
				set_variable = {
					name = wealth
					value = scope:wealth
				}
				set_variable = {
					name = book_content_quality
					value = scope:book_content_quality
				}
			}
			# All if/else_if a removed and is replaced by this effect
			GH_add_artifact_modifier_from_list_effect = yes # We retrieve the modifiers that existed on the artifact before		
		}
	}

	#To clean out variables
	if = {
		limit = {
			exists = scope:author
		}
		scope:author = {
			if = {
				limit = {
					has_variable = book_subject
				}
				remove_variable = book_subject
			}
			if = {
				limit = {
					has_variable = book_topic
				}
				remove_variable = book_topic
			}
			if = {
				limit = {
					has_variable = book_topic_category
				}
				remove_variable = book_topic_category
			}
			if = {
				limit = {
					has_variable = book_content_quality
				}
				remove_variable = book_content_quality
			}
		}
	}
	scope:owner = {
		if = {
			limit = { has_variable = book_artifact_subject }
			remove_variable = book_artifact_subject
		}
		if = {
			limit = { has_variable = book_artifact_topic }
			remove_variable = book_artifact_topic
		}
		if = {
			limit = { has_variable = book_topic_category }
			remove_variable = book_topic_category
		}
		if = {
			limit = {
				has_variable = book_topic_category
			}
			remove_variable = book_topic_category
		}
		if = {
			limit = {
				has_variable = book_content_quality
			}
			remove_variable = book_content_quality
		}
	}

	if = {
		limit = {
			exists = scope:should_be_trinket 
		}
		clear_saved_scope = should_be_trinket 	
	}
}
