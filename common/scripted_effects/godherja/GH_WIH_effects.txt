influence_phase_impact_level_injector = {
	#We remove it before adding anything to the variable
	remove_variable ?= great_struggle_influence_impact_level
	
	
	set_variable = {
		name = tempory_great_struggle_influence_impact_level
		value = $STRUGGLE_PHASE$_impact_level_value
	}
	#Make it slightly more randomized
	random_list = {
		10 = { 
			change_variable = {
				name = tempory_great_struggle_influence_impact_level
				add = -1
			}
		}
		50 = { 
			change_variable = {
				name = tempory_great_struggle_influence_impact_level
				add = 0
			}
		}
		20 = { 
			change_variable = {
				name = tempory_great_struggle_influence_impact_level
				add = 1
			}
		}
		15 = { 
			change_variable = {
				name = tempory_great_struggle_influence_impact_level
				add = 2
			}
		}
		3 = { 
			change_variable = {
				name = tempory_great_struggle_influence_impact_level
				add = 3
			}
		}
		2 = { 
			change_variable = {
				name = tempory_great_struggle_influence_impact_level
				add = 4
			}
		}
	}
	
	if = {
		limit = { var:tempory_great_struggle_influence_impact_level <= 1 }
		set_variable = {
			name = great_struggle_influence_impact_level
			value = flag:weak
		}
		while = {
			count = 1
			activate_great_struggle_wih_injector_effect = { STRUGGLE_PHASE = $STRUGGLE_PHASE$ }
		}
	}
	else_if = {
		limit = { 
			var:tempory_great_struggle_influence_impact_level > 1 
			var:tempory_great_struggle_influence_impact_level <= 3 
		}
		set_variable = {
			name = great_struggle_influence_impact_level
			value = flag:moderate
		}
		while = {
			count = 3
			activate_great_struggle_wih_injector_effect = { STRUGGLE_PHASE = $STRUGGLE_PHASE$ }
		}
	}
	else_if = {
		limit = { 
			var:tempory_great_struggle_influence_impact_level > 3 
			var:tempory_great_struggle_influence_impact_level <= 7 
		}
		set_variable = {
			name = great_struggle_influence_impact_level
			value = flag:strong
		}
		while = {
			count = 5
			activate_great_struggle_wih_injector_effect = { STRUGGLE_PHASE = $STRUGGLE_PHASE$ }
		}
	}
	else_if = {
		limit = { 
			var:tempory_great_struggle_influence_impact_level > 7 
		}
		set_variable = {
			name = great_struggle_influence_impact_level
			value = flag:overwhelming
		}
		while = {
			count = 8
			activate_great_struggle_wih_injector_effect = { STRUGGLE_PHASE = $STRUGGLE_PHASE$ }
		}
	}
}
