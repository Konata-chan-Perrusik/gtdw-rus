personal_inspiration_wealth_value = { # Artifact Wealth of a Personal Inspiration
	value = 0
	add = { # Character Rank
		if = { # Adventurers
			limit = { story_owner = { has_government = landless_adventurer_government } }
			add = 10
		}
		else_if = { # Noble Family House Head (Landless)
			limit = {
				story_owner = {
					is_landless_ruler = yes
					any_held_title = { is_noble_family_title = yes }
				}
			}
			add = 20
		}
		else = { # Landed Rulers
			story_owner = {
				switch = {
					trigger = primary_title.tier
					tier_county = { value = 10 }
					tier_duchy = { value = 20 }
					tier_kingdom = { value = 30 }
					tier_empire = { value = 40 }
					tier_hegemony = { value = 50 }
				}
			}
		}
		desc = personal_inspiration_wealth_value_rank
		format = "WEALTH_VALUE_FORMAT"
	}
	if = { # Family Estate Location Development
		limit = { story_owner = { any_held_title = { is_noble_family_title = yes } } }
		add = {
			value = story_owner.domicile.domicile_location.county.development_level
			divide = 2
			desc = personal_inspiration_wealth_value_development_level_desc
			format = "WEALTH_VALUE_FORMAT"
		}
	}
	else = { # Capital County Development
		add = {
			value = story_owner.capital_county.development_level
			divide = 2
			desc = personal_inspiration_wealth_value_development_level_desc
			format = "WEALTH_VALUE_FORMAT"
		}
	}
	add = { # Purchased materials
		value = possible_personal_inspiration_actions_materials_wealth_value
		multiply = var:possible_personal_inspiration_actions_materials_count
		desc = personal_inspiration_wealth_value_materials_desc
		format = "WEALTH_VALUE_FORMAT"
	}
	add = { # From events
		value = var:personal_inspiration_wealth
		desc = personal_inspiration_wealth_value_story_desc
		format = "WEALTH_VALUE_FORMAT"
	}
}
