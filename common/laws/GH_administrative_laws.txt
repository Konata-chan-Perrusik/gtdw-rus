	cumulative = yes

	otachedian_province_war_law_0 = {
		modifier = {
		}
		flag = uses_otachedian_bureaucracy

		can_keep = {
			realm_law_uses_otachedian_bureaucracy = yes
			trigger_if = {
				limit = {
					top_liege != this
				}
				liege = { has_realm_law = otachedian_province_war_law_0 }
			}
		}

		can_pass = {
			trigger_if = {
				limit = {
					top_liege != this
					liege = { has_realm_law = otachedian_province_war_law_0 }
				}
				custom_tooltip = {
					text = imperial_bureaucracy_pending_law_change
					always = no
				}
			}
		}

		on_pass = {
			## Set cooldown
			set_variable = {
				name = admin_law_cooldown
				years = @admin_law_cooldown_value
			}
		}
	}
