# Characters get divorced in any way
# root = character initiating the divorce
# scope:spouse = the divorced spouse
# scope:reason = flag:script or flag:faith
on_divorce = { # Will not fire from historical divorces
	on_actions = {
		GH_on_divorce
	}
}

GH_on_divorce = { # Will not fire from historical divorces
	effect = {
		if = {
			limit = {
				NOT = { faith.religion = scope:spouse.faith.religion }
				any_character_struggle = {
					activate_struggle_catalyst_secondary_character_involvement_involved_or_aligned_trigger = {
						CATALYST = catalyst_interfaith_marriage
						CHAR = scope:spouse
					}
				}
			}
			every_character_struggle = {
				limit = {
					activate_struggle_catalyst_secondary_character_involvement_involved_or_aligned_trigger = {
						CATALYST = catalyst_interfaith_marriage
						CHAR = scope:spouse
					}
				}
				activate_struggle_catalyst = {
					catalyst = catalyst_interfaith_marriage
					character = scope:spouse
				}
			}
		}
	}
}
