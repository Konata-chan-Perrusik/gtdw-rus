GH_WIH_activate_struggle_catalyst_secondary_character_involvement_either_trigger = {
	phase_has_catalyst = $CATALYST$
	save_temporary_scope_as = struggle_temp
	$CHAR$ = {
		any_character_struggle = {
			OR = { 
				this = struggle:gh_war_in_heaven_struggle 
				this = struggle:gh_eastern_war_in_heaven_struggle
				this = struggle:gh_far_eastern_war_in_heaven_struggle
				this = struggle:gh_western_war_in_heaven_struggle
				#this = struggle:gh_far_western_war_in_heaven_struggle
			}
			this = scope:struggle_temp
		}
	}
}

gh_wih_not_benefited_from_previous_phase_injector_trigger = { 
	trigger_if = {
		limit = { has_religion = religion:wardenism_religion }
		NOT = { global_var:gh_$REGION$_war_in_heaven_struggle_previous_phase ?= flag:struggle_wih_phase_wardenite_dominance }
	}
	trigger_else_if = {
		limit = { has_religion = religion:ritualist_religion }
		NOT = { global_var:gh_$REGION$_war_in_heaven_struggle_previous_phase ?= flag:struggle_wih_phase_ritualist_dominance }
	}
	trigger_else_if = {
		limit = { 
			any_character_struggle = {
				OR = { 
					is_struggle_type = gh_war_in_heaven_struggle
					is_struggle_type = gh_eastern_war_in_heaven_struggle
				}
			}
			has_religion = religion:agionism_religion 
		}
		NOT = { global_var:gh_$REGION$_war_in_heaven_struggle_previous_phase ?= flag:struggle_wih_phase_agionist_dominance }
	}
	trigger_else = {}
}

gh_does_this_player_care_about_the_war_in_heaven = {
	NOT = { this = root }
	OR = {
		location = {
			OR = { # Is in region
				geographical_region = world_sarradon
			}
		}
		scope:saved_struggle = { is_faith_involved_in_struggle = prev.faith }
	}
}

	any_county_in_region = {
		region = $STRUGGLE_REGION$
		percent >= 0.85
		holder.top_liege ?= root
	}
}

activate_struggle_catalyst_secondary_character_involvement_involved_or_aligned_trigger = {
	phase_has_catalyst = $CATALYST$
	save_temporary_scope_as = struggle_temp
	$CHAR$ = {
		OR = {
			any_character_struggle = {
				involvement = involved
				this = scope:struggle_temp
			}
			AND = {
				OR = {
					has_trait = gh_wardenite_aligned
					has_trait = gh_ritualist_aligned
					has_trait = gh_agionist_aligned 
				} 
				any_character_struggle = { 
					involvement = interloper
					this = scope:struggle_temp 
				}
			}
		}
	}
}
