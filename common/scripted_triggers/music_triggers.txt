# Various music triggers

should_use_asian_music_trigger = {
	OR = {
		AND = {
			culture = {
				has_building_gfx = steppe_building_gfx
			}
			OR = {
				culture ?= { has_cultural_pillar = heritage_sarminian }
				culture ?= { has_oltenic_heritage_pillar_trigger = yes }
			}
			capital_province ?= {
				OR = {
					geographical_region = world_asia_north_east
					geographical_region = world_steppe_east
				}
			}
		}
		culture = {
			OR = {
				has_building_gfx = chinese_building_gfx
				has_building_gfx = indian_building_gfx
				has_building_gfx = southeast_asian_building_gfx
				has_building_gfx = tibetan_building_gfx
				has_building_gfx = japanese_building_gfx
			}
		}
		#religion = religion:buddhism_religion
		#religion = religion:hinduism_religion
		any_character_situation = { situation_type = dynastic_cycle }
	}
}

should_use_chinese_music_trigger = {
	OR = {
		culture = { has_cultural_pillar = heritage_easterner }
		culture = { any_parent_culture_or_above = { has_cultural_pillar = heritage_easterner } }
		any_character_situation = { situation_type = dynastic_cycle }
	}
}

should_use_nomadic_music_trigger = {
	OR = {
		culture ?= { has_cultural_pillar = heritage_sarminian }
		culture ?= { has_oltenic_heritage_pillar_trigger = yes }
		AND = {
			exists = root.capital_county
			government_has_flag = government_is_nomadic
			situation:the_sarradonian_desert = {
				any_situation_sub_region = {
					situation_sub_region_has_county = root.capital_county
				}
			}
		}
		AND = {
			exists = root.capital_county
			government_has_flag = government_is_nomadic
			situation:the_great_steppe = {
				any_situation_sub_region = {
					situation_sub_region_has_county = root.capital_county
				}
			}
		}
	}
}

should_use_iberian_music_trigger = {
	OR = {
		culture = {
			has_shattered_coast_heritage_pillar_trigger = yes
		}
		culture = {
			any_parent_culture_or_above = { has_shattered_coast_heritage_pillar_trigger = yes }
		}
		AND = {
			exists = capital_province
			capital_province = { geographical_region = world_europe_west_iberia }
		}
	}
}

should_use_iranian_music_trigger = {
	OR = {
		culture = { has_sarradonian_heritage_pillar_trigger = yes }
		AND = {
			religion = religion:wardenism_religion
			NAND = {
				government_has_flag = government_is_nomadic
				situation:the_sarradonian_desert ?= {
					any_situation_sub_region = {
						situation_sub_region_has_county = root.capital_county
					}
				}
			}
		}
		religion = religion:amsari_religion
	}
}

should_use_specifically_persian_music_trigger = {

		AND = {
			exists = capital_province
			capital_province = { geographical_region = world_ws_amsari }
		}
	}
}
