# Locate Prism
# by hihi
# Shows who is currently holding the abyssal prism. 
# It's already possible to find the prism by looking through all characters, but this should be less frustrating.

find_prism_decision = {  
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_ck2_devil_worshiper_sacrifice.dds"
	}
	cooldown = { years = 5 }
	desc = find_prism_decision_desc
	decision_group_type = redland_cults
	selection_tooltip = find_prism_decision_tooltip
	
	is_shown = {
		any_artifact = { 
			has_variable = abyssal_prism
		}
		NOT = {
			any_character_artifact = {
				has_variable = abyssal_prism
			}
		}
		has_faith = faith:redland_cults
	}	
	is_valid = {
		has_faith = faith:redland_cults
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
		piety >= 1000
	}
	effect = {
		save_scope_as = prism_searcher_1

		every_living_character = {
			limit = {
				any_character_artifact = {
					has_variable = abyssal_prism
				}
			}
			save_scope_as = current_prism_owner
		}
		trigger_event = redland_cults_events.0001
		custom_tooltip = find_prism_decision_tooltip
	}

	cost = {
		piety = 1000
	}
	ai_check_interval = 0
	ai_potential = {
		always = no
	}
	ai_will_do = {
		base = 0
	}
}

## Pray for Proliferative Growth
## Give away extra provinces to get under domain limit (redlanders can only have 1 county)
redland_cults_proliferation_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_ck2_devil_worshiper_sacrifice.dds"
	}
	decision_group_type = redland_cults
	ai_check_interval = 12

	is_shown = {
		is_landed = yes
		is_redlander = yes
		domain_size > domain_limit
	}

	is_valid = {
		is_landed = yes
		is_redlander = yes
		domain_size > domain_limit
	}

	cost = {
		piety = 0
	}

	effect = {
		every_held_title = {
			limit = {
				tier = tier_county
				NOT = { this = root.capital_county }
			}
			save_scope_as = extra_county
		}
		redland_cults_proliferation_scripted_effect = {
			CULTOWNER = root
			EXTRATITLE = scope:extra_county
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
