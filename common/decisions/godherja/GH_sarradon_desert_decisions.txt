establish_almusid_culture_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp2_decision_struggle_conciliation.dds"
	}
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 120
		kingdom = 120
		empire = 0
		hegemony = 0
	}

	desc = establish_almusid_culture_decision_desc
	selection_tooltip = establish_almusid_culture_decision_tooltip

	is_shown = {
		game_start_date >= 6050.1.1
		culture = culture:utyalid
		any_sub_realm_county = {
			target_is_de_jure_liege_or_above = title:d_almudir
			OR = {
				culture = culture:dawasid
				culture = culture:madikirib
			}
		}
		NOT = {	
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_formed_almusid_decision
			}
		}
	}

	is_valid = {
		trigger_if = {
			limit = { game_start_date >= 6298.1.1 }
			custom_tooltip = {
				text = gh_ten_years_past_for_settling_nomad_requirement_tt
				current_date >= 6308.1.1		
			}
		}
		trigger_else = {}
		culture = culture:utyalid
		completely_controls = title:d_almudir
	}

	is_valid_showing_failures_only = {
		is_at_war = no
		is_imprisoned = no
	}

	effect = {
		hidden_effect = {
			# Generate a list of counties we want to flip to the new culture.
			title:d_almudir = {
				every_in_de_jure_hierarchy = {
					limit = { tier = tier_county}
					add_to_list = almusid_counties
				}
			}
			every_sub_realm_title = {
				limit = {
					tier = tier_county
				}
				if = {
					limit = {
						title_province = { geographical_region = custom_almudir_and_adjacent }
						OR = {
							culture = culture:dawasid
							culture = culture:madikirib
							culture = culture:utyalid
						}
					}
					add_to_list = almusid_counties
				}
			}

			# Perform the culture flip.
			every_in_list = {
				list = almusid_counties
				merge_cultures_in_region_effect = {
					OLD_CULTURE_1 = culture:utyalid
					OLD_CULTURE_2 = culture:dawasid
					NEW_CULTURE = culture:almusid
					REGION = custom_almudir_and_adjacent
				}
			}
			every_in_list = {
				list = almusid_counties
				merge_cultures_in_region_effect = {
					OLD_CULTURE_1 = culture:utyalid
					OLD_CULTURE_2 = culture:madikirib
					NEW_CULTURE = culture:almusid
					REGION = custom_almudir_and_adjacent
				}
			}

			# Send all players in the game a notification event about the flip.
			save_scope_as = almusid_founder
			culture:utyalid = {save_scope_as = old_culture_1}
			culture:dawasid = {save_scope_as = old_culture_2}
			culture:madikirib = {save_scope_as = old_culture_3}
			culture:almusid = {
				reset_culture_creation_date = yes
				get_all_innovations_from = scope:old_culture_1
				get_all_innovations_from = scope:old_culture_2
				get_all_innovations_from = scope:old_culture_3
				save_scope_as = new_culture
			}
			every_player = {
				if = {
					limit = {
						this != root
						OR = {
							culture = culture:utyalid
							culture = culture:dawasid
							culture = culture:madikirib
						}
						capital_province ?= { geographical_region = custom_almudir_and_adjacent }
					}
					trigger_event = sarradon_decisions.3001
				}
				else_if = {
					limit = { this = root }
					trigger_event = sarradon_decisions.3002
				}
				else_if = {
					limit = {
						capital_province ?= { geographical_region = world_sarradon }
					}
					trigger_event = sarradon_decisions.3003
				}
			}
		}

		# Player changes culture
		set_culture = culture:almusid
		add_character_flag = converted_culture_this_lifetime
		hidden_effect = {
			every_in_list = {
				list = almusid_counties
				limit = {
					holder.capital_county = { is_in_list = almusid_counties }
				}
				holder = {
					add_to_list = mass_melting_pot_list
					every_courtier = { add_to_list = mass_melting_pot_list }
					every_courtier_away = { add_to_list = mass_melting_pot_list }
				}
				every_in_list = {
					list = mass_melting_pot_list
					limit = {
						OR = {
							culture = culture:utyalid
							culture = culture:dawasid
							culture = culture:madikirib
						}
					}
					set_culture = culture:almusid
				}
			}
		}

		# Summary of above hidden effects to inform players of the decision's impact.
		show_as_tooltip = {
			custom_tooltip = establish_almusid_culture_decision_effect.tt
			culture:almusid = {
				get_all_innovations_from = scope:old_culture_1
				#get_all_innovations_from = scope:old_culture_2
				get_all_innovations_from = scope:old_culture_3
			}
		}

		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:flag_formed_almusid_decision
		}

		dynasty = { add_dynasty_prestige = 200 }
	}

	cost = {
		prestige = {
			value = minor_prestige_value
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 0

		modifier = {
			add = 100
			any_sub_realm_county = {
				percent >= 0.5
				OR = {
					culture = culture:dawasid
					culture = culture:madikirib
				}
			}
		}
	}
}
