	ai_check_interval = 60

	selection_tooltip = gh_align_in_war_in_heaven_decision_tooltip

	is_shown = {
		#Standard filter checks.
		is_playable_character = yes
		exists = dynasty

		any_character_struggle = {
			involvement = interloper

				is_struggle_type = gh_war_in_heaven_struggle

				is_struggle_type = gh_far_eastern_war_in_heaven_struggle

		}
		#Isn't already aligned to one side
		NOR = {
			has_trait = gh_wardenite_aligned
			has_trait = gh_ritualist_aligned
		}
	}

	is_valid = {
		is_independent_ruler = yes
		any_character_struggle = {
			involvement = interloper
			NOR = { 
				is_struggle_phase = struggle_wih_central_sarradon_phase_detente
				is_struggle_phase = struggle_wih_eastern_sarradon_phase_detente 
				is_struggle_phase = struggle_wih_far_eastern_sarradon_phase_detente
				is_struggle_phase = struggle_wih_western_sarradon_phase_detente 
				#is_struggle_phase = struggle_wih_far_western_sarradon_phase_detente		
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		custom_tooltip = {
			text = recently_left_side_in_war_in_heaven_tt
			NOT = { has_variable = recently_left_side_in_war_in_heaven }		
		}
	}
	
	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller

			icon = "gfx/interface/icons/decision_items/gh_wardenite_aligned.dds"
			ai_chance = {
				value = 0
				if = {
					limit = {
						any_character_struggle = {
							involvement = interloper
							OR = { 
								is_struggle_phase = struggle_wih_central_sarradon_phase_wardenite_dominance
								is_struggle_phase = struggle_wih_eastern_sarradon_phase_wardenite_dominance

				}

					limit = {
						any_relation = {
							type = rival
							has_religion = religion:wardenism_religion
						}	
					}
					add = -50

				}
				if = {
					limit = {
						any_neighboring_top_liege_realm_owner = {
							is_confederation_member = no
							has_religion = religion:wardenism_religion

			localization = align_with_ritualist_item
			icon = "gfx/interface/icons/decision_items/gh_ritualist_aligned.dds"
			ai_chance = {
				value = 0
				if = {
					limit = {
						any_character_struggle = {
							involvement = interloper
							OR = { 
								is_struggle_phase = struggle_wih_central_sarradon_phase_ritualist_dominance
								is_struggle_phase = struggle_wih_eastern_sarradon_phase_ritualist_dominance
								is_struggle_phase = struggle_wih_far_eastern_sarradon_phase_ritualist_dominance

							}
						}	
					}

				}
				if = {

							type = rival
							has_religion = religion:ritualist_religion

							has_religion = religion:ritualist_religion
							NOT = { is_allied_to = root }
							confederation_worthy_foe_strength_ratio_value_root <= 0.25
						}					
					}
					add = 60
				}
			}
		}
		item = { # Side with the Agionist
			value = align_with_agionist_item
			is_valid = {}
			current_description = align_with_agionist_item_desc
			localization = align_with_agionist_item
			icon = "gfx/interface/icons/decision_items/gh_agionist_aligned.dds"
			ai_chance = {
				value = -50
				if = {
					limit = {
						any_character_struggle = {
							involvement = interloper
							OR = { 
								is_struggle_phase = struggle_wih_central_sarradon_phase_agionist_dominance
								is_struggle_phase = struggle_wih_eastern_sarradon_phase_agionist_dominance

						}	
					}
					add = 10
				}
				if = {
					limit = {
						any_relation = {
							type = rival
							has_religion = religion:agionism_religion
						}					
					}
					add = -50
				}
				if = {
					limit = {
						any_neighboring_top_liege_realm_owner = {
							is_confederation_member = no
							has_religion = religion:agionism_religion
							NOT = { is_allied_to = root }
							confederation_worthy_foe_strength_ratio_value_root <= 0.25
						}

	effect = {
		if = {
			limit = { scope:align_with_wardenite_item = yes }

			custom_tooltip = align_with_wardenite_item_tooltip
			custom_tooltip = align_with_wardenite_item_enemy_tooltip
		}
		else_if = {
			limit = { scope:align_with_ritualist_item = yes }
			add_trait = gh_ritualist_aligned
			custom_tooltip = align_with_ritualist_item_tooltip
			custom_tooltip = align_with_ritualist_item_enemy_tooltip

			limit = { scope:align_with_agionist_item = yes }
			add_trait = gh_ritualist_aligned
			custom_tooltip = align_with_ritualist_item_tooltip
			custom_tooltip = align_with_ritualist_item_enemy_tooltip
		}
		add_prestige = medium_prestige_gain	
		stress_impact = {
			craven = medium_stress_impact_gain
			brave = minor_stress_impact_loss
			ambitious = miniscule_stress_impact_loss

		prestige = 200

	}

	ai_potential = {
		any_character_struggle = {
			involvement = interloper
			is_struggle_type = gh_war_in_heaven_struggle
		}
		short_term_gold >= medium_gold_value
	}

	ai_will_do = {
		base = -20
		modifier = {
			add = 60
			any_neighboring_top_liege_realm_owner = {
				is_confederation_member = no
				OR = {
					has_religion = religion:wardenism_religion

				}	
				NOT = { is_allied_to = root }
				confederation_worthy_foe_strength_ratio_value_root <= 0.25
			}
		}
		modifier = {
			add = 10
			has_trait = ambitious
		}
		modifier = {
			add = -30
			has_trait = craven
		}
		modifier = {
			add = -30

		}
		modifier = {
			add = -500
			this = character:hudi_yugaug_6
			this = character:mutsartan_wardenism_15
			this = character:tapuso_selothaan_1
		}

gh_central_sarradon_wardenite_victory_ending_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp3/fp3_decision_supremacy.dds"
	}
	extra_picture = "gfx/interface/illustrations/struggle_decision_buttons/fp3_decision_supremacy.dds"
	decision_group_type = major
	is_invisible = yes

	desc = gh_central_sarradon_wardenite_victory_ending_decision_desc
	selection_tooltip = gh_central_sarradon_wardenite_victory_ending_decision_tooltip

	sort_order = 80

	is_shown = {
		is_landless_adventurer = no
	}

	is_valid = {
		trigger_if = {
			limit = {
				is_ai = no
				calc_true_if = {
					amount = 1
					exists = struggle:gh_war_in_heaven_struggle
					exists = struggle:gh_eastern_war_in_heaven_struggle
					exists = struggle:gh_far_eastern_war_in_heaven_struggle
					exists = struggle:gh_western_war_in_heaven_struggle
					#exists = struggle:gh_far_western_war_in_heaven_struggle
				}
			}
			custom_tooltip = {
				text = gh_great_struggle_must_be_ended_in_window
				has_variable = can_end_struggle_player
			}
		}
		trigger_else = {}
		is_independent_ruler = yes
		
		OR = {
			custom_tooltip = {
				text = gh_wardenite_victory_ending_decision_correct_phase_tt
				any_character_struggle = {
					is_struggle_type = gh_war_in_heaven_struggle
					is_struggle_phase = struggle_wih_central_sarradon_phase_wardenite_dominance
				}
			}
			## Completely control Central Sarradon for 5 years
			AND = {
				completely_controls_region = GH_central_sarradon_struggle_region
				primary_title =  {
					tier >= tier_kingdom
					title_held_years >= 5
				}
			}
		}

		custom_tooltip = { #
			text = struggle_central_sarradon_ending_decision_kingdom_tt
			any_held_title = { # 1st kingdom
				tier = tier_kingdom
				de_jure_liege = title:h_sarradon
				root = { completely_controls = prev }
				save_temporary_scope_as = sarradon_kingdom
			}
			any_held_title = { # 2nd kingdom
				tier = tier_kingdom
				de_jure_liege = title:h_sarradon
				trigger_if = {
					limit = { exists = scope:sarradon_kingdom }
					NOT = { this = scope:sarradon_kingdom }
				}
				root = { completely_controls = prev }
				save_temporary_scope_as = sarradon_kingdom_2
			}
		}
			
		custom_tooltip = {
			text = gh_wih_struggle_ending_decision_correct_involvement_tt
			any_character_struggle = {
				involvement = involved
				is_struggle_type = gh_war_in_heaven_struggle
			}
		}
		#Control at least two of the main kingdoms
		calc_true_if = {
			amount >= 2
			completely_controls = title:d_first_kingdom
			completely_controls = title:d_tell_habja
			completely_controls = title:d_dozzan
			completely_controls = title:d_beid_faluh
			completely_controls = title:d_sagecrowns
			completely_controls = title:d_thronelands
			completely_controls = title:d_rifeh
			completely_controls = title:d_aversaria_peripontica
		}
		OR = {
			AND = {
				custom_tooltip = {
					text = gh_wardenite_victory_ending_decision_no_ritualist_rulers_tt
					NOT = {
						struggle:gh_war_in_heaven_struggle = {		
							any_involved_ruler = {
								NOT = { this = root }
								is_independent_ruler = yes	
								exists = primary_title
								has_religion = religion:ritualist_religion
							}
						}
					}
				}
				custom_tooltip = {
					text = gh_central_sarradon_ending_struggle_counties_mostly_wardenite_religion_tt
					any_county_in_region = {
						region = GH_central_sarradon_struggle_region
						percent >= 0.90
						faith.religion = religion:wardenism_religion
					}
				}
			}
			AND = {
				custom_tooltip = {
					text = gh_central_sarradon_ending_struggle_counties_sixty_wardenite_faith_tt
					any_county_in_region = {
						region = GH_central_sarradon_struggle_region
						percent >= 0.60
						faith = root.faith
					}
				}
				custom_tooltip = {
					text = gh_central_sarradon_ending_struggle_control_most_counties_tt
					any_county_in_region = {
						region = GH_central_sarradon_struggle_region
						percent >= 0.75
						holder.top_liege ?= root
					}
				}
				custom_tooltip = {
					text = gh_central_sarradon_ending_struggle_less_ritualist_rulers_tt
					NOT = {
						any_county_in_region = {
							region = GH_central_sarradon_struggle_region
							percent >= 0.10
							holder.top_liege ?= { faith.religion = religion:ritualist_religion }
						}
					}
				}
			}		
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
	}


	effect = {
		custom_tooltip = {
			text = gh_central_sarradon_wardenite_victory_ending_decision_effect_tooltip
			gh_great_struggle_ended_regional_struggle_info_effect = {
				STRUGGLE_NAME = gh_region_central_war_in_heaven_struggle
				REASON = gh_wih_ending_wardenite_victory
				REASON_DESCRIPTION = gh_wih_ending_wardenite_victory_desc
				STRUGGLE_ENDER = root
			}
		}
		
		if = {
			limit = { has_dlc_feature = legends }
			legend_seed_struggle_ending_effect = {
				ENDER = root
				STRUGGLE = gh_war_in_heaven_struggle
				STRUGGLE_ENDING = wardenite
			}
		}
		
		##### Major Effects #####
		show_as_tooltip = {
			dynasty = { add_dynasty_prestige = 5000 }
			give_nickname = nick_muqtih_al_alayyiun
		}
		# Sarradon is available!
		custom_tooltip = gh_struggle_can_create_empire_of_sarradon_tt
		custom_description_no_bullet = { text = fp2_struggle_house_tt }
		# Boost to faith conversion
		custom_tooltip = gh_struggle_wih_ending_conversion_tt
		# Damage opinion with other faiths
		custom_tooltip = gh_struggle_wih_ending_opinion_tt
		custom_tooltip = gh_struggle_wih_ending_ritualist_opinion_negative_tt
		# Choose one modifier
		custom_description_no_bullet = { text = gh_struggle_wih_ending_list_tt }

		show_as_tooltip = {
			stress_impact = {
				humble = medium_stress_impact_gain
				cynical = medium_stress_impact_gain
				zealous = medium_stress_impact_loss
			}
		}
		
		# Trigger a player facing event as a coda
		trigger_event = gh_war_in_heaven_struggle.0100
	}
	
	cost = {}

	ai_check_interval = 120

	ai_potential = { always = yes }

	ai_will_do = { base = 100 }
}

gh_eastern_sarradon_wardenite_victory_ending_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp3/fp3_decision_supremacy.dds"
	}
	extra_picture = "gfx/interface/illustrations/struggle_decision_buttons/fp3_decision_supremacy.dds"
	decision_group_type = major
	is_invisible = yes

	desc = gh_eastern_sarradon_wardenite_victory_ending_decision_desc
	selection_tooltip = gh_eastern_sarradon_wardenite_victory_ending_decision_tooltip

	sort_order = 80

	is_shown = {
		is_landless_adventurer = no
	}

	is_valid = {
		trigger_if = {
			limit = {
				is_ai = no
				calc_true_if = {
					amount = 1
					exists = struggle:gh_war_in_heaven_struggle
					exists = struggle:gh_eastern_war_in_heaven_struggle
					exists = struggle:gh_far_eastern_war_in_heaven_struggle
					exists = struggle:gh_western_war_in_heaven_struggle
					#exists = struggle:gh_far_western_war_in_heaven_struggle
				}
			}
			custom_tooltip = {
				text = gh_great_struggle_must_be_ended_in_window
				has_variable = can_end_struggle_player
			}
		}
		trigger_else = {}
		is_independent_ruler = yes
		
		OR = {
			custom_tooltip = {
				text = gh_wardenite_victory_ending_decision_correct_phase_tt
				any_character_struggle = {
					is_struggle_type = gh_eastern_war_in_heaven_struggle
					is_struggle_phase = struggle_wih_eastern_sarradon_phase_wardenite_dominance
				}
			}
			## Completely control Central Sarradon for 5 years
			AND = {
				completely_controls_region = GH_eastern_sarradon_struggle_region
				primary_title =  {
					tier >= tier_kingdom
					title_held_years >= 5
				}
			}
		}

		custom_tooltip = { #
			text = struggle_eastern_sarradon_ending_decision_kingdom_tt
			any_held_title = { # 1st kingdom
				tier = tier_kingdom
				de_jure_liege = title:h_sarradon
				root = { completely_controls = prev }
				save_temporary_scope_as = sarradon_kingdom
			}
			any_held_title = { # 2nd kingdom
				tier = tier_kingdom
				de_jure_liege = title:h_sarradon
				trigger_if = {
					limit = { exists = scope:sarradon_kingdom }
					NOT = { this = scope:sarradon_kingdom }
				}
				root = { completely_controls = prev }
				save_temporary_scope_as = sarradon_kingdom_2
			}
		}
			
		custom_tooltip = {
			text = gh_wih_struggle_ending_decision_correct_involvement_tt
			any_character_struggle = {
				involvement = involved
				is_struggle_type = gh_eastern_war_in_heaven_struggle
			}
		}
		#Control at least two of the main kingdoms
		calc_true_if = {
			amount >= 2
			completely_controls = title:d_valley_of_the_djinn_kings
			completely_controls = title:d_al_miwf
			completely_controls = title:d_khekech
			completely_controls = title:d_khenem_delta
			completely_controls = title:d_forlorn_coast
			completely_controls = title:d_isle_of_visions
			completely_controls = title:d_western_dreamrend_isles
		}
		OR = {
			AND = {
				custom_tooltip = {
					text = gh_wardenite_victory_ending_decision_no_ritualist_rulers_tt
					NOT = {
						struggle:gh_eastern_war_in_heaven_struggle = {		
							any_involved_ruler = {
								NOT = { this = root }
								is_independent_ruler = yes	
								exists = primary_title
								has_religion = religion:ritualist_religion
							}
						}
					}
				}
				custom_tooltip = {
					text = gh_eastern_sarradon_ending_struggle_counties_mostly_wardenite_religion_tt
					any_county_in_region = {
						region = GH_eastern_sarradon_struggle_region
						percent >= 0.90
						faith.religion = religion:wardenism_religion
					}
				}
			}
			AND = {
				custom_tooltip = {
					text = gh_eastern_sarradon_ending_struggle_counties_sixty_wardenite_faith_tt
					any_county_in_region = {
						region = GH_eastern_sarradon_struggle_region
						percent >= 0.60
						faith = root.faith
					}
				}
				custom_tooltip = {
					text = gh_eastern_sarradon_ending_struggle_control_most_counties_tt
					any_county_in_region = {
						region = GH_eastern_sarradon_struggle_region
						percent >= 0.75
						holder.top_liege ?= root
					}
				}
				custom_tooltip = {
					text = gh_eastern_sarradon_ending_struggle_less_ritualist_rulers_tt
					NOT = {
						any_county_in_region = {
							region = GH_eastern_sarradon_struggle_region
							percent >= 0.10
							holder.top_liege ?= { faith.religion = religion:ritualist_religion }
						}
					}
				}
			}		
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
	}


	effect = {
		custom_tooltip = {
			text = gh_eastern_sarradon_wardenite_victory_ending_decision_effect_tooltip
			gh_great_struggle_ended_regional_struggle_info_effect = {
				STRUGGLE_NAME = gh_region_eastern_war_in_heaven_struggle
				REASON = gh_wih_ending_wardenite_victory
				REASON_DESCRIPTION = gh_wih_ending_wardenite_victory_desc
				STRUGGLE_ENDER = root
			}
		}
		
		if = {
			limit = { has_dlc_feature = legends }
			legend_seed_struggle_ending_effect = {
				ENDER = root
				STRUGGLE = gh_eastern_war_in_heaven_struggle
				STRUGGLE_ENDING = wardenite
			}
		}
		
		##### Major Effects #####
		show_as_tooltip = {
			dynasty = { add_dynasty_prestige = 5000 }
			give_nickname = nick_zayr_al_misrak
		}

		custom_description_no_bullet = { text = fp2_struggle_house_tt }
		# Boost to faith conversion
		custom_tooltip = gh_struggle_wih_ending_conversion_tt
		# Damage opinion with other faiths
		custom_tooltip = gh_struggle_wih_ending_opinion_tt
		custom_tooltip = gh_struggle_wih_ending_ritualist_opinion_negative_tt
		# Choose one modifier
		custom_description_no_bullet = { text = gh_struggle_wih_ending_list_tt }

		show_as_tooltip = {
			stress_impact = {
				humble = medium_stress_impact_gain
				cynical = medium_stress_impact_gain
				zealous = medium_stress_impact_loss
			}
		}
		
		# Trigger a player facing event as a coda
		trigger_event = gh_war_in_heaven_struggle.0100
	}
	
	cost = {}

	ai_check_interval = 120

	ai_potential = { always = yes }

	ai_will_do = { base = 100 }
}

	}

	is_valid = {
		NOR = {
			has_religion = religion:wardenism_religion

					NOR = {
						is_struggle_phase = struggle_wih_eastern_sarradon_phase_ritualist_dominance
						is_struggle_phase = struggle_wih_eastern_sarradon_phase_wardenite_dominance

			}

			text = gh_eastern_sarradon_ending_struggle_half_ritualist_or_wardenite_tt
			NOT = {
				any_county_in_region = {
					region = GH_eastern_sarradon_struggle_region
					percent >= 0.6

						faith.religion = religion:wardenism_religion
						faith.religion = religion:ritualist_religion
					}
				}
			}
		}

			text = gh_eastern_sarradon_generic_victory_ending_decision_decision_region_tt

gh_eastern_sarradon_agionist_victory_ending_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp2_decision_struggle_opening.dds"
	}
	extra_picture = "gfx/interface/illustrations/struggle_decision_buttons/GH_WIH_agionist_ending.dds"
	decision_group_type = major
	is_invisible = yes

	desc = gh_eastern_sarradon_agionist_victory_ending_decision_desc
	selection_tooltip = gh_eastern_sarradon_agionist_victory_ending_decision_tooltip

	sort_order = 80

	is_shown = {
		is_landless_adventurer = no
	}

	is_valid = {
		has_religion = religion:agionism_religion
		
		trigger_if = {
			limit = {
				is_ai = no
				calc_true_if = {
					amount = 1
					exists = struggle:gh_war_in_heaven_struggle
					exists = struggle:gh_eastern_war_in_heaven_struggle
					exists = struggle:gh_far_eastern_war_in_heaven_struggle
					exists = struggle:gh_western_war_in_heaven_struggle
					#exists = struggle:gh_far_western_war_in_heaven_struggle
				}
			}
			custom_tooltip = {
				text = gh_great_struggle_must_be_ended_in_window
				has_variable = can_end_struggle_player
			}
		}
		trigger_else = {}
		is_independent_ruler = yes

		OR = {
			custom_tooltip = {
				text = gh_agionist_victory_ending_decision_correct_phase_tt
				any_character_struggle = {
					is_struggle_type = gh_eastern_war_in_heaven_struggle
					is_struggle_phase = struggle_wih_eastern_sarradon_phase_agionist_dominance
				}
			}
			## Completely control Central Sarradon for 5 years
			AND = {
				completely_controls_region = GH_eastern_sarradon_struggle_region
				primary_title =  {
					tier >= tier_kingdom
					title_held_years >= 5
				}
			}
		}
		
		custom_tooltip = { #
			text = struggle_eastern_sarradon_ending_decision_kingdom_tt
			any_held_title = { # 1st kingdom
				tier = tier_kingdom
				de_jure_liege = title:h_sarradon
				root = { completely_controls = prev }
				save_temporary_scope_as = sarradon_kingdom
			}
			any_held_title = { # 2nd kingdom
				tier = tier_kingdom
				de_jure_liege = title:h_sarradon
				trigger_if = {
					limit = { exists = scope:sarradon_kingdom }
					NOT = { this = scope:sarradon_kingdom }
				}
				root = { completely_controls = prev }
				save_temporary_scope_as = sarradon_kingdom_2
			}
		}
		custom_tooltip = {
			text = gh_eastern_sarradon_ending_struggle_fourty_ritualist_or_wardenite_tt
			NOT = {
				any_county_in_region = {
					region = GH_eastern_sarradon_struggle_region
					percent >= 0.4
					OR = {
						faith.religion = religion:wardenism_religion
						faith.religion = religion:ritualist_religion
					}
				}
			}
		}
		#Control at least two of the main kingdoms
		calc_true_if = {
			amount >= 2
			completely_controls = title:d_valley_of_the_djinn_kings
			completely_controls = title:d_al_miwf
			completely_controls = title:d_khekech
			completely_controls = title:d_khenem_delta
			completely_controls = title:d_forlorn_coast
			completely_controls = title:d_isle_of_visions
			completely_controls = title:d_western_dreamrend_isles
		}
		
		OR = {
			AND = {	
				custom_tooltip = {
					text = gh_generic_victory_ending_decision_no_ritualist_or_wardenite_rulers_tt
					NOT = {
						struggle:gh_eastern_war_in_heaven_struggle = {		
							any_involved_ruler = {
								NOT = { this = root }
								is_independent_ruler = yes	
								exists = primary_title
								OR = {
									faith.religion = religion:ritualist_religion 
									faith.religion = religion:wardenism_religion 
								}
							}
						}
					}
				}			
				OR = {
					custom_tooltip = {
						text = gh_eastern_sarradon_ending_struggle_control_most_counties_tt
						any_county_in_region = {
							region = GH_eastern_sarradon_struggle_region
							percent >= 0.60
							holder.top_liege ?= root
						}
					}
					custom_tooltip = {
						text = gh_eastern_sarradon_ending_struggle_agionist_controlled_eighty_counties_tt
						any_county_in_region = {
							region = GH_eastern_sarradon_struggle_region
							percent >= 0.80
							holder.top_liege ?= { has_religion = religion:agionism_religion }
						}
					}
				}
			}
			AND = {
				custom_tooltip = {
					text = gh_eastern_sarradon_ending_struggle_agionist_religion_tt
					any_county_in_region = {
						region = GH_eastern_sarradon_struggle_region
						percent >= 0.60
						faith = root.faith
					}
				}
				custom_tooltip = {
					text = gh_eastern_sarradon_ending_struggle_less_ritualist_or_wardenite_rulers_tt
					NOT = {
						any_county_in_region = {
							region = GH_eastern_sarradon_struggle_region
							percent >= 0.10
							holder.top_liege ?= { 
								OR = {
									faith.religion = religion:ritualist_religion 
									faith.religion = religion:wardenism_religion 
								}
							}
						}
					}
				}				
				OR = {
					custom_tooltip = {
						text = gh_eastern_sarradon_ending_struggle_control_sixty_percent_counties_tt
						any_county_in_region = {
							region = GH_eastern_sarradon_struggle_region
							percent >= 0.60
							holder.top_liege ?= root
						}
					}
					custom_tooltip = {
						text = gh_eastern_sarradon_ending_struggle_agionist_controlled_eighty_counties_tt
						any_county_in_region = {
							region = GH_eastern_sarradon_struggle_region
							percent >= 0.80
							holder.top_liege ?= { has_religion = religion:agionism_religion }
						}
					}
				}

			}		
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
	}
	
	effect = {
		custom_tooltip = {
			text = gh_eastern_sarradon_agionist_victory_ending_decision_effect_tooltip
			gh_great_struggle_ended_regional_struggle_info_effect = {
				STRUGGLE_NAME = gh_region_eastern_war_in_heaven_struggle
				REASON = gh_wih_ending_agionist_victory
				REASON_DESCRIPTION = gh_wih_ending_agionist_victory_desc
				STRUGGLE_ENDER = root
			}
		}
		
		if = {
			limit = { has_dlc_feature = legends }
			legend_seed_struggle_ending_effect = {
				ENDER = root
				STRUGGLE = gh_eastern_war_in_heaven_struggle
				STRUGGLE_ENDING = agionist
			}
		}
		
		##### Major Effects #####
		show_as_tooltip = {
			dynasty = { add_dynasty_prestige = 6000 }
			add_prestige = 3000
			give_nickname = nick_the_crusader
		}
		
		#Struggle Available in the Region
		custom_tooltip = gh_wih_ending_struggle_conquest_available_tt
		custom_description_no_bullet = { text = fp2_struggle_house_tt }
		# Damage opinion with other faiths
		custom_tooltip = gh_struggle_wih_ending_ritualist_opinion_negative_tt
		custom_tooltip = gh_struggle_wih_ending_wardenite_opinion_negative_tt

		show_as_tooltip = {
			if = {
				limit = { exists = house }
				house = {
					add_house_modifier = {
						modifier = gh_house_ended_eastern_sarradon_wih_struggle_patron_affinity 
					}
				}
			}
			stress_impact = {
				humble = medium_stress_impact_gain
				cynical = medium_stress_impact_gain
				zealous = medium_stress_impact_loss
			}
		}
		
		# Trigger a player facing event as a coda
		trigger_event = gh_war_in_heaven_struggle.0400
		
		custom_tooltip = {
			text = gh_eastern_sarradon_ending_struggle_patron_saint_affinity
			set_variable = gh_ended_eastern_sarradon_wih_struggle_patron_affinity
		}
		
		custom_tooltip = {
			text = gh_eastern_sarradon_ending_struggle_patron_saint_affinity_house
			set_variable = gh_ended_eastern_sarradon_wih_struggle_patron_affinity
		}
	}
	
	cost = {}

	ai_check_interval = 120

	ai_potential = { always = yes }

	ai_will_do = { base = 100 }
}
