gh_great_struggle_wardenite_ending_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp3/fp3_decision_supremacy.dds"

	extra_picture = "gfx/interface/illustrations/struggle_decision_buttons/gh_great_struggle_wardenite_ending_decision.dds"

	desc = gh_great_struggle_wardenite_ending_decision_desc
	selection_tooltip = gh_great_struggle_wardenite_ending_decision_tooltip

	sort_order = 80

	is_valid = {
		any_character_struggle = {
			OR = { 
				is_struggle_type = gh_war_in_heaven_struggle
				is_struggle_type = gh_eastern_war_in_heaven_struggle
				is_struggle_type = gh_far_eastern_war_in_heaven_struggle

				#is_struggle_type = gh_far_western_war_in_heaven_struggle	
			}
		}
		prestige_level >= high_prestige_level
		trigger_if = {
			limit = {
				calc_true_if = {
					amount = 1
					exists = struggle:gh_war_in_heaven_struggle
					exists = struggle:gh_eastern_war_in_heaven_struggle
					exists = struggle:gh_far_eastern_war_in_heaven_struggle
					exists = struggle:gh_western_war_in_heaven_struggle
					#exists = struggle:gh_far_western_war_in_heaven_struggle
				}
			}
			trigger_if = {
				limit = { exists = struggle:gh_war_in_heaven_struggle }
				can_execute_decision = gh_central_sarradon_wardenite_victory_ending_decision
			}
			trigger_else_if = {
				limit = { exists = struggle:gh_eastern_war_in_heaven_struggle }
				can_execute_decision = gh_eastern_sarradon_wardenite_victory_ending_decision
			}
			trigger_else_if = {
				limit = { exists = struggle:gh_far_eastern_war_in_heaven_struggle }
				can_execute_decision = gh_far_eastern_sarradon_wardenite_victory_ending_decision
			}
			trigger_else_if = {
				limit = { exists = struggle:gh_western_war_in_heaven_struggle }
				can_execute_decision = gh_western_sarradon_wardenite_victory_ending_decision
			}
			#trigger_else_if = {
			#	limit = { exists = struggle:gh_far_western_war_in_heaven_struggle }
			#	can_execute_decision = gh_far_western_sarradon_wardenite_victory_ending_decision
			#}
			trigger_else = { always = no }
		}
		trigger_else = {
			custom_tooltip = {
				text = gh_great_stuggle_ending_multiple_struggle_exists_tt
				always = no
			}
		}
	}
	
	is_valid_showing_failures_only = {
		is_available_adult = yes
	}


	effect = {
		show_as_tooltip = {
			dynasty = { add_dynasty_prestige = 10000 }
			add_prestige = 7000
			give_nickname = nick_ma_dun_al_ira
		}

		custom_description_no_bullet = { text = fp2_struggle_house_tt }
		custom_tooltip = gh_struggle_wih_ending_ritualist_opinion_negative_tt


		show_as_tooltip = {
			stress_impact = {
				humble = medium_stress_impact_gain
				cynical = medium_stress_impact_gain
				zealous = medium_stress_impact_loss
			}
		}
		custom_tooltip = {	
			text = gh_great_stuggle_ending_all_struggles_end_tt		
			if = { 
				limit = { exists = struggle:gh_war_in_heaven_struggle }
				struggle:gh_war_in_heaven_struggle = { end_struggle = yes }
			}
			if = { 
				limit = { exists = struggle:gh_eastern_war_in_heaven_struggle }
				struggle:gh_eastern_war_in_heaven_struggle = { end_struggle = yes }
			}
			if = { 
				limit = { exists = struggle:gh_far_eastern_war_in_heaven_struggle }
				struggle:gh_far_eastern_war_in_heaven_struggle = { end_struggle = yes }
			}
			if = { 
				limit = { exists = struggle:gh_western_war_in_heaven_struggle }
				struggle:gh_western_war_in_heaven_struggle = { end_struggle = yes }
			}
			#if = { 
			#	limit = { exists = struggle:gh_far_western_war_in_heaven_struggle }
			#	struggle:gh_far_western_war_in_heaven_struggle = { end_struggle = yes }
			#}
		}
		custom_tooltip = {	
			text = gh_great_stuggle_ending_inconclusive_tt
			remove_global_variable = wih_great_struggle
		}
	}
	cost = {
		piety = major_piety_value
        prestige = {
            value = major_prestige_value
        }
	}

	ai_check_interval = 0

	ai_will_do = { base = 100 }
}
