favour_the_whispering_rite_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_levy_outcasts.dds" }
	decision_group_type = sjalvolki

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 120
		empire = 120
		hegemony = 120
	}

	desc = favour_the_whispering_rite_decision_desc
	selection_tooltip = favour_the_whispering_rite_decision_tooltip

	is_shown = {
		is_playable_character = yes
		exists = dynasty
		faith.religion = religion:kardawai_religion
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:favoured_the_whispering_rite
			}
		}
	}

	is_valid = {
		OR = {
			exists = struggle:rite_wars_south_struggle
			exists = struggle:ritewars_struggle
		}
		OR = {
			piety_level >= 3
			custom_description = {
				text = favour_the_whispering_rite_decision.you_have_x_or_more_traits_virtuous_to_whispering_rite
				num_virtuous_traits = {
					value >= 2
					target = faith:whispering_rite
				}
			}
		}
		highest_held_title_tier >= tier_duchy
		prestige_level >= 3
		top_liege = this
	}

	is_valid_showing_failures_only = {
		is_available_at_peace_adult = yes
	}

	effect = {
		gain_holy_legend_seed_tooltip_effect = yes
		save_scope_as = religious_leader
		faith = { save_scope_as = old_faith }
		faith:whispering_rite = { save_scope_as = new_faith }

		show_as_tooltip = {
			form_the_whispering_rite_decision_generic_effects_scripted_effect = yes
			random_list = {
				100 = {
					show_chance = no
					desc = favour_the_whispering_rite_decision.path_default
					form_the_whispering_rite_decision_default_path_scripted_effect = yes
				}
				100 = {
					show_chance = no
					desc = favour_the_whispering_rite_decision.path_cannibalism_curtailed
					form_the_whispering_rite_decision_cannibalism_curtailed_path_scripted_effect = yes
				}
			}
		}
		
		trigger_event = sjalvolki_reformation.0406
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:favoured_the_whispering_rite
			}
		}
	}

	cost = {
		piety = 1000
		prestige = 1000
	}

	ai_potential = {
		highest_held_title_tier >= tier_duchy
	}

	ai_will_do = {
		base = 0

		modifier = {
			add = 25
			has_trait = arrogant
		}
		modifier = {
			add = 25
			has_trait = stubborn
		}
		modifier = {
			add = 75
			any_sub_realm_county = {
				percent >= 0.51
				OR = {
					culture = culture:kardwens
					culture = culture:sjalvolki
				}
			}
		}
	}
